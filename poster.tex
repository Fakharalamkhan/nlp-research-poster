% ===================== 1 HYPOTHESIS =====================
\sbox{\boxA}{\begin{minipage}[t][\colH][t]{\wA}
\sechead{1}{Hypothesis}
\begin{minipage}[t][\bodyH][s]{\linewidth}\fsBody\color{textdark}\raggedright
\sub{Problem}
Intrinsic self-critique often turns right answers into wrong ones~[1,\,2].\par
\vfill
\sub{Idea}
Revise only if the confidence in $A_0$ is low:\par\vspace{4mm}
{\centering{\fsBig\bfseries\color{accent} revise $\Leftrightarrow S(A_0) < \tau$}\par}
\vfill
\sub{RQs and hypotheses}
\rqh{1}{Does a gate beat never- and always-revise?}{Yes, it beats both.}
\thinrule
\rqh{2}{Which signal best finds wrong answers?}{Self-consistency.}
\thinrule
\rqh{3}{How large can the gain of a gate be?}{It is capped by the fix rate~$f$:}
\vspace{4mm}
{\centering\fsBig$\begin{aligned}\mathrm{Acc} &= a + (1{-}a)\,f\,t\\ &\quad - a\,b\,u\end{aligned}$\par}\vspace{4mm}
{\fsCap
\begin{tabular}{@{}l@{\hspace{4mm}}>{\raggedright\arraybackslash}p{\dimexpr\linewidth-12mm\relax}@{}}
$a$ & initial accuracy\\
$f$ & fix rate\\
$t$ & share of wrong answers revised\\
$b$ & break rate\\
$u$ & share of right answers revised
\end{tabular}\par}\vspace{3mm}
Ceiling on the gain: {\bfseries\color{accent}$(1{-}a)\,f$}\par
\vfill
\sub{Prior work}
\bul{IoE: keep or revise inside one prompt~[3]}
\bul{Confidence vs.\ critique ability~[4]}
\bul{P(True) self-verification~[5]}
\bul{Self-consistency by sampling~[6]}
\bul{New here: explicit threshold gates, tuned on held-out data}
\end{minipage}
\end{minipage}}

\sechead{2}{Methodology}
\begin{minipage}[t][\bodyH][s]{\linewidth}\fsBody\color{textdark}\raggedright
\sub{Pipeline}
{\centering
\begin{tikzpicture}[
  pstep/.style={draw=navy, line width=0.8mm, rounded corners=1mm, fill=white, text width=112mm,

\node[pdec, below=7mm of s] (d) {$S(A_0) < \tau$\,?};
\node[pout, anchor=north west] (rev) at ([yshift=-10mm]d.south -| q.west) {Revise:\\ output $A_1$};

\draw[parr] (d.east) -| node[plab, right, pos=0.75] {no} (keep.north);
\end{tikzpicture}\par}
\vfill
\sub{Setup and why}
\bul{Qwen2.5-7B-Instruct~[9]: open, fits 2$\times$ NVIDIA T4 with vLLM~[10]}
\bul{GSM8K~[7] (math), HotpotQA~[8] (multi-hop QA): answers checked automatically}
\bul{100 dev / 400 test each: $\tau$ never tuned on test}
\vfill
\sub{Signals}
One from each family in prior work:\par\vspace{2mm}
\bul{Verbalised confidence $\Sv$}
\bul{Self-consistency $\Ssc$, $N{=}10$ samples~[6]}
\bul{P(True) $\Sf$~[5]}
\vfill
\sub{Baselines, metrics}
\bul{Never, always, majority vote, IoE~[3]; oracle as upper bound}
\bul{Majority vote as revision step: a step that does fix errors}
\bul{$W{\to}R$ / $R{\to}W$ instead of net accuracy}
\bul{AUROC, token cost; paired exact McNemar test}
\vfill
\sub{Code and data}
{\fsTab\href{\posterrepo/tree/main/experiments}{github.com/\allowbreak Fakharalamkhan/\allowbreak nlp-research-poster}\par}
{\fsCap folder \texttt{experiments/}: code, prompts, data splits, raw outputs\par}
\end{minipage}
\end{minipage}}
