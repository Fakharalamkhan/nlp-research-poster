% ===================== 1 HYPOTHESIS =====================
\sbox{\boxA}{\begin{minipage}[t]{\wA}
\begin{tcolorbox}[section, title={\secnum{1}\hspace{5mm}Hypothesis}]
\begin{minipage}[t][\tcbtextheight][s]{\linewidth}\fsBody\color{textdark}\raggedright
\sub{Problem}
\bul{Intrinsic self-critique often turns right answers into wrong ones~[1,\,2].}
\vfill
\sub{Idea}
Revise only if the confidence in $A_0$ is low:\par\vspace{3mm}
\begin{tcolorbox}[panel, halign=center]
{\fsBig\bfseries\color{accenttext} revise $\Leftrightarrow S(A_0) < \tau$}
\end{tcolorbox}
\vfill
\sub{RQs and hypotheses}
\begin{tcolorbox}[card]{\color{headerblue}\bfseries RQ1}\enspace Does a gate beat never- and always-revise?\par\vspace{1mm}
{\color{accenttext}\bfseries H1}\enspace Yes, it beats both.\end{tcolorbox}
\begin{tcolorbox}[card]{\color{headerblue}\bfseries RQ2}\enspace Which signal best finds wrong answers?\par\vspace{1mm}
{\color{accenttext}\bfseries H2}\enspace Self-consistency.\end{tcolorbox}
\begin{tcolorbox}[card, after skip=0pt]{\color{headerblue}\bfseries RQ3}\enspace How large can the gain of a gate be?\par\vspace{1mm}
{\color{accenttext}\bfseries H3}\enspace It is capped by the fix rate~$f$:\par\vspace{4mm}
  {\fsBig$\begin{aligned}\mathrm{Acc} &= a + (1{-}a)\,f\,t\\ &\quad - a\,b\,u\end{aligned}$\par}\vspace{4mm}
  {\fsCap
  \begin{tabular}{@{}l@{\hspace{4mm}}>{\raggedright\arraybackslash}p{\dimexpr\linewidth-12mm\relax}@{}}
  $a$ & initial accuracy\\
  $f$ & fix rate\\
  $t$ & share of wrong answers revised\\
  $b$ & break rate\\
  $u$ & share of right answers revised
  \end{tabular}\par}\vspace{3mm}
  Ceiling on the gain: {\bfseries\color{accenttext}$(1{-}a)\,f$}
\end{tcolorbox}
\vfill
\sub{Prior work}
\bul{IoE: keep or revise inside one prompt~[3]}
\bul{Confidence vs.\ critique ability~[4]}
\bul{P(True) self-verification~[5]}
\bul{Self-consistency by sampling~[6]}
\bul{New here: explicit threshold gates, tuned on held-out data}
\end{minipage}
\end{tcolorbox}
\end{minipage}}

\begin{minipage}[t][\tcbtextheight][s]{\linewidth}\fsBody\color{textdark}\raggedright
\sub{Pipeline}
{\centering
\begin{tikzpicture}[
  pstep/.style={draw=headerblue, line width=1mm, rounded corners=4mm, fill=white, text width=112mm,
    align=center, minimum height=15mm, font=\fsBody},

\node[pdec, below=7mm of s] (d) {$S(A_0) < \tau$\,?};
\node[pout, anchor=north west] (rev) at ([yshift=-10mm]d.south -| q.west) {Revise:\\ output $A_1$};
\node[pout, anchor=north east] (keep) at ([yshift=-10mm]d.south -| q.east) {Keep $A_0$};
\draw[parr] (q) -- (a0);
\draw[parr] (a0) -- (s);

\draw[parr] (d.west) -| node[plab, left, pos=0.75] {yes} (rev.north);
\draw[parr] (d.east) -| node[plab, right, pos=0.75] {no} (keep.north);
\end{tikzpicture}\par}
\vfill
\sub{Setup and why}
\bul{Qwen2.5-7B-Instruct~[9]: open, fits 2$\times$ NVIDIA T4 with vLLM~[10]}
\bul{GSM8K~[7] (math), HotpotQA~[8] (multi-hop QA): answers checked automatically}
\bul{100 dev / 400 test each: $\tau$ never tuned on test}
\vfill
\sub{Signals}
One from each family in prior work:\par\vspace{2mm}
\bul{Verbalised confidence $\Sv$}
\bul{Self-consistency $\Ssc$, $N{=}10$ samples~[6]}
\bul{P(True) $\Sf$~[5]}
\vfill
\sub{Baselines, metrics}
\bul{Never, always, majority vote, IoE~[3]; oracle as upper bound}
\bul{Majority vote as revision step: a step that does fix errors}
\bul{$W{\to}R$ / $R{\to}W$ instead of net accuracy}
\bul{AUROC, token cost; paired exact McNemar test}
\vfill
